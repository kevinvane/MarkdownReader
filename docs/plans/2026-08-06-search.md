**步骤 3: 编写搜索执行函数**

在 `renderer.js` 末尾追加下列函数。核心匹配逻辑按行执行，返回行级结果。

```js
function buildRegExp(query, useRegex, caseSensitive) {
  if (useRegex) return new RegExp(query, caseSensitive ? 'g' : 'gi');
  return new RegExp(escapeRegExp(query), caseSensitive ? 'g' : 'gi');
}

function escapeRegExp(str) {
  return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

function runSearch(query, useRegex, caseSensitive) {
  clearSearchHighlights();
  const input = $('search-count');           // 该 span 用于计数
  if (!query) {
    searchResults = [];
    searchTotalHits = 0;
    searchActiveIndex = -1;
    renderResults(null, '输入关键词开始搜索');
    return;
  }
  let re;
  try {
    re = buildSearchExp(query, useRegex, caseSensitive);
  } catch (e) {
    $('search-input').classList.add('is-error');
    renderResults([], '正则表达式错误：' + e.message);
    return;
  }
  $('search-input').classList.remove('is-error');
  const lines = rawMarkdown.split('\n');
  const results = [];
  let totalHits = 0;
  for (let i = 0; i < lines.length; i++) {
    re.lastIndex = 0;
    const m = lines[i].match(re);
    if (m) {
      const count = m.length;
      totalHits += count;
      results.push({ lineNo: i, lineText: lines[i], hitCount: count });
      if (results.length >= 500) { truncationNotice = true; break; }
    }
  }
  searchResults = results;
  searchTotalHits = totalHits;
  searchResults.length = Math.min(searchResults.length, 500);
  searchActiveIndex = searchResults.length ? 0 : -1;
  renderResults(results, results.length ? null : '未找到匹配项');
  if (searchActiveIndex >= 0) highlightSeResult(searchActiveIndex);
}
```

> 实现时以**命中总数**填状态栏（`searchTotalHits`），列表按**行**展示（每行含 `hitCount`）。截断计数：当结果被 500 条截断时，状态栏附带"结果过多，仅显示前 500 条"提示。

高亮片段伪逻辑（写入 `renderResults` 中一个工具函数 `highlightLine(line, re)`）：
```js
function highlightLine(line, re) {
  re.lastIndex = 0;
  const parts = [];
  let last = 0;
  let m;
  while ((m = re.exec(line)) && m[0]) {
    if (m.index > last) parts.push(escapeHtml(line.slice(last, m.index)));
    parts.push('<mark>' + escapeHtml(m[0]) + '</mark>');
    last = m.index + m[0].length;
  }
  parts.push(escapeHtml(line.slice(last)));
  return parts.join('');
}
```

### Task 4: 快捷键 + 事件绑定（`renderer.js`）

**Files:**
- Modify: `renderer.js`

**步骤 1: 面板开关、防抖、Enter 跳转**

在 `renderer.js` 末尾追加事件绑定函数 `initSearchPanel()`，并在脚本底部调用。

关键逻辑：
- **Ctrl+F / Cmd+F**（`document.addEventListener('keydown')`）：`preventDefault()` 阻止浏览器原生查找，切换面板；展开时聚焦并全选输入框已有文本。
- **Esc**：关闭面板（`hidden`）+ `clearSearchState()`（清输入、结果、高亮）。
- **输入框 input 事件**：`clearTimeout(searchDebounceId)` 后 `setTimeout(300ms)` 调 `runSearch()`。
- **输入框 Enter / Shift+Enter**：在结果间跳转（正向/反向），同步主内容区滚动。
- **[Aa] / [.*] 按钮 click**：切换 `aria-pressed`，`clearTimeout` 立即重新 `runSearch()`。

```js
function initSearchPanel() {
  const panel = $('search-panel');
  const input = $('search-input');
  const btnCase = $('search-btn-case');
  const btnRegex = $('search-btn-regex');

  document.addEventListener('keydown', (e) => {
    if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'f') {
      e.preventDefault();
      if (panel.hidden) { openSearchPanel(); }
      else { closeSearchPanel(); }
      return;
    }
    if (e.key === 'Escape' && !panel.hidden) { closeSearchPanel(); }
  });

  input.addEventListener('input', () => {
    clearTimeout(searchDebounceId);
    searchDebounceId = setTimeout(() => runSearch(), 300);
  });

  input.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') {
      e.preventDefault();
      if (!searchResults.length) return;
      moveSelection(e.shiftKey ? -1 : 1);
    }
  });

  btnCase.addEventListener('click', () => {
    btnCase.setAttribute('aria-pressed', String(btnCase.getAttribute('aria-pressed') !== 'true'));
    btnCase.textContent = btnCase.getAttribute('aria-pressed') === 'true' ? 'A' : 'Aa';
    rerunSearchNow();
  });

  btnRegex.addEventListener('click', () => {
    btnRegex.setAttribute('aria-pressed', btnRegex.getAttribute('aria-pressed') !== 'true');
    rerunSearchNow();
  });
}
```

### Task 5: 行号 → 内容区跳转映射（`renderer.js` + `index.html` 高亮容器）

**Files:**

**步骤 1: 借助 markdown-it `token.map` 建立映射**

在 `renderMarkdown` 中，`md.render` 之前无法直接拿到 tokens；用 `md.parse` 获得 tokens，记录"源码起始行号 → 生成的块元素"。为最小侵入，在 `renderMarkdown` 中手动 `md.parse(content)`，用每个开块 token 的 `map[0]` 建立 `lineNo → blockIndex` 数组，然后交给 `buildLineMap` 填充 `isCenterMap` 的映射。

```js
function buildLineMap(content) {
  isLineMap = [];
  if (!content) return;
  const tokens = md.parse(content, {});
  // 只关心块级 token（map 存在），记录其起始源码行
  tokens.forEach(t => {
    if (t.map && t.map[0] >= 0) {
      // token.map[0] 为源码起始行号；渲染时生成的元素位于内容区同一顺序
      isLineMap.push({ line: t.map[0], index: isLineMap.length });
    }
  });
}
```

**步骤 2: 跳转实现 `jumpToLine(lineNo)`**

核心：遍历 `#content` 下的块级元素（`p`、`h*`、`pre`、`table`、`li`、`ul` 等），按 `buildLineMap` 里记录的索引顺序找到距离 `lineNo` 最近、且 `startLine <= lineNo` 的最后一个块，对其 `scrollIntoView({behavior:'smooth', block:'start'})` 并临时加 `.search-result-mark` 高亮（自动清除）。

```js
function jumpToLine(lineNo, searchRegExp) {
  const contentEl = $('content');
  const blocks = contentEl.querySelectorAll(':scope > *');   // 顶层块
  // 找到第一个"起始行 >= lineNo"的块，向前回退一个即包含该行
  let target = blocks[0];
  for (const b of blocks) {
    target = b;
  }
  target.scrollIntoView({ behavior: 'smooth', block: 'start' });
  flashHighlight(target);
}

function flashHighlight(el) {
  el.classList.add('search-result-mark');
  clearTimeout((el._flashTimer));
  el._flashTimer = setTimeout(() => el.classList.remove('search-result-mark'), 1200);
}
```

> **实现要点（契合需求 5.4）**：由于 `token.map[0]` 是行的**近似**映射，`.` 要求"匹配所在位置进入视口且被识别"即可，不做逐像素精确。若需更精确可进一步用 `#content` 内子元素行数估算，但本期以块级近似为准。
